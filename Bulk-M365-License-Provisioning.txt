# CSV must contain a UserPrincipalName column

# Import users from CSV
$Users = Import-Csv "CSV PATH"

# Get the license SKU
$license = Get-MgSubscribedSku | Where-Object { $_.SkuPartNumber -eq "<MICROSOFT_365_LICENSE_SKU_PART_NUMBER>" }

if (-not $license) {
    Write-Host "License not found. Please check the SKU part number." -ForegroundColor Red
    return
}

foreach ($user in $Users) {
    $upn = $user.UserPrincipalName
    try {
        $userObject = Get-MgUser -Filter "UserPrincipalName eq '$upn'"
        if ($userObject) {
            $userLicense = Get-MgUserLicenseDetail -UserId $userObject.Id | Where-Object { $_.SkuId -eq $license.SkuId }
            if ($userLicense) {
                Write-Host "License $($license.SkuPartNumber) is already assigned to $($userObject.DisplayName)" -ForegroundColor Yellow
            }
            else {
                Write-Host "Assigning license $($license.SkuPartNumber) to $($userObject.DisplayName)" -ForegroundColor Green
                Set-MgUserLicense -UserId $userObject.Id -AddLicenses @{SkuId = $license.SkuId } -RemoveLicenses @() -ErrorAction Stop
            }
        }
        else {
            Write-Host "User $upn not found." -ForegroundColor Red
        }
    }
   catch {
    Write-Host ("Error processing " + $upn + ": " + $_.Exception.Message) -ForegroundColor Red
    continue
}

}
